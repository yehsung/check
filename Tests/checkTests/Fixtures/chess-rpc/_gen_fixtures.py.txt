def psql(sql, user="migrator"):
    p = subprocess.run([PSQL, "-X", "-q", "-At", "-v", "ON_ERROR_STOP=1", "-U", user, "-d", "postgres"],
                       input=sql, capture_output=True, text=True, env=env)
    if p.returncode != 0:
        sys.stderr.write(sql + "\n--- stderr ---\n" + p.stderr)
        raise SystemExit(f"psql failed rc={p.returncode}")
    return p.stdout


def admin(sql):
    psql("set role postgres;\nbegin;\n" + sql + "\ncommit;\n")


def scalar(sql):
    return psql("set role postgres;\nselect '@@' || (" + sql + ")::text;\n").strip()[2:]


def call(caller, rpc_sql, setup="", end="rollback", name=None, note="", role="authenticated", flagged=None):
    login = "select harness.login({}, '{}') \\g /dev/null\n".format(
        "null::uuid" if caller is None else "'%s'" % caller, role)
    sql = "set role postgres;\nbegin;\n" + setup + "\n" + login
    sql += "select '@@' || (%s)::text;\n%s;\n" % (rpc_sql, end)
    out = psql(sql)
    lines = [l for l in out.splitlines() if l.startswith("@@")]
    assert len(lines) == 1, out
    obj = json.loads(lines[0][2:], object_pairs_hook=OrderedDict)
    if name:
        with open(os.path.join(OUT, name + ".json"), "w", encoding="utf-8") as fh:
            json.dump(obj, fh, ensure_ascii=False, indent=2)
            fh.write("\n")
        manifest[name] = OrderedDict(caller=caller, caller_label=LABEL.get(caller), rpc=rpc_sql,
                                     setup=setup.strip() or None, end=end, flagged=flagged,
                                     note=note or None)
        extra = ""
        if "match" in obj:
            extra = " ply=%s turn=%s" % (obj["match"].get("ply_count"), obj["match"].get("turn"))
        print("%-46s status=%-18s%s" % (name, obj.get("status"), extra))
    return obj


# ── 판 심기(표에 직접 — RPC 로 만들면 루비·초인종·푸시에 묶인다) ─────────────────────────────
def seed(mid, white, black, fen, ply, wms, bms, wg, bg, ago_ms, stake=5, status="active",
         result=None, reason=None, winner=None, finished_ago="0 seconds"):
    cols = dict(
        id="'%s'" % mid, challenger="'%s'" % white, opponent="'%s'" % black, stake=str(stake),
        status="'%s'" % status, white="'%s'" % white, black="'%s'" % black, fen="'%s'" % fen,
        ply_count=str(ply), white_ms_left=str(wms), black_ms_left=str(bms),
        white_grace_ms=str(wg), black_grace_ms=str(bg),
        turn_started_at="clock_timestamp() - make_interval(secs => %s)" % (ago_ms / 1000.0),
        created_at="clock_timestamp() - interval '1 hour'",
        invite_expires_at="clock_timestamp() - interval '59 minutes'",
        accepted_at="clock_timestamp() - interval '59 minutes'",
        result="null" if result is None else "'%s'" % result,
        end_reason="null" if reason is None else "'%s'" % reason,
        winner="null" if winner is None else "'%s'" % winner,
        finished_at="null" if status != "finished" else "clock_timestamp() - interval '%s'" % finished_ago)
    admin("delete from public.chess_matches where id = '%s';\ninsert into public.chess_matches(%s) values (%s);"
          % (mid, ", ".join(cols.keys()), ", ".join(cols.values())))
    # ★ 심은 판의 **출발 국면**을 매니페스트에 남긴다 — 수가 하나뿐인 픽스처(외통·승격·불충분)에서
    #   Swift 가 "직전 국면에 그 수를 적용한 결과" 를 혼자 계산할 수 있어야 한다.
    seeds[mid] = OrderedDict(fen=fen, ply=ply, white=white, black=black, status=status,
                             white_ms_left=wms, black_ms_left=bms, stake=stake)
    return mid

         note="상대가 이미 제안했다 — 덮지 않는다")
    call(black, "public.chess_respond_draw(1, '%s', false)" % m1, end="commit",
         name="chess_respond_draw__decline_ok", note="제안만 지운다(draw_offer_by null)")

    # 관전 — m1 은 진행 중
    call(C, "public.chess_watch(1, '%s', 0)" % m1, name="chess_watch__ok_active",
         note="진행 중 — white_user/black_user · my_color·legal_moves·ruby_balance 없음")
    call(C, "public.chess_watch(1, '%s', 4)" % m1, name="chess_watch__ok_since4",
         note="관전도 since 가 배타다")
    call(white, "public.chess_watch(1, '%s', 0)" % m1, name="chess_watch__ok_as_player",
         note="대국자도 관전 RPC 를 부를 수 있다(쓰기가 없어 깃발이 떨어지지 않는다)")
    call(HIDDEN, "public.chess_watch(1, '%s', 0)" % m1, name="chess_watch__not_found_hidden",
         note="숨김 계정은 일반 사용자 판을 관전할 수 없다(same_visibility 두 조건)")

    # m1 종료 — 기권
    call(black, "public.chess_resign(1, '%s')" % m1, end="commit", name="chess_resign__ok",
         note="흑 기권 → white_win/resign · 승자 +2×판돈")
    call(black, "public.chess_resign(1, '%s')" % m1, name="chess_resign__not_active", note="멱등")
    call(white, "public.chess_watch(1, '%s', 0)" % m1, name="chess_watch__ok_tail",
         note="끝난 판은 chess_watch_tail_seconds()=180초 꼬리만 보인다")
    call(white, "public.chess_move(1, '%s', 6, 'd2d4')" % m1, name="chess_move__not_active")

    # ── 6. 종국 넷 + 불충분 기물 + 합의 ────────────────────────────────────────────────────
    # 외통: §11 ② 와 같은 국면(백 Ra8#)
    seed(M_MATE, A, B, "6k1/5ppp/8/8/8/8/8/R5K1 w - - 0 1", 0, 300000, 300000, 2000, 2000, 100)
    call(A, "public.chess_move(1, '%s', 0, 'a1a8')" % M_MATE, end="commit",
         name="chess_move__end_checkmate", note="Ra8# → checkmate:white · 백 +2×판돈")

    # 3회 반복: Nf3 Nf6 Ng1 Ng8 Nf3 Nf6 Ng1 Ng8 — 마지막 수에서 시작 국면이 3회가 된다
    seed(M_THREE, A, B, "rnbqkbnr/pppppppp/8/8/8/8/PPPPPPPP/RNBQKBNR w KQkq - 0 1", 0,
         300000, 300000, 2000, 2000, 100)
    rep = ["g1f3", "g8f6", "f3g1", "f6g8", "g1f3", "g8f6", "f3g1", "f6g8"]
    ids["threefold_line"] = rep
    ids["threefold_start_fen"] = scalar("public.chess_start_fen()")
    for index, uci in enumerate(rep):
        mover = A if index % 2 == 0 else B
        last = index == len(rep) - 1
        r = call(mover, "public.chess_move(1, '%s', %d, '%s')" % (M_THREE, index, uci), end="commit",
                 name="chess_move__end_threefold" if last else None,
                 note="8수째에 시작 국면이 **3회**가 된다(chess__rep_count: 수 기록 + 시작 국면 1)" if last else "")
        assert r["status"] == "ok", (uci, r)
        if last:
            assert r["repetition"] == 3 and r["outcome"] == "threefold_repetition", r

    # 불충분 기물: Bxb5 뒤 작은 말이 판 전체에 하나(왕+비숍 vs 왕)
    seed(M_INSUF, A, B, "4k3/8/8/1b6/8/8/8/4KB2 w - - 0 40", 78, 300000, 300000, 2000, 2000, 100)
    call(A, "public.chess_move(1, '%s', 78, 'f1b5')" % M_INSUF, end="commit",
         name="chess_move__end_insufficient", note="Bxb5 → 왕+비숍 vs 왕 = 사국(판 전체를 본다)")

    # 시간패: 백이 e4 를 **실제로** 두고(그래서 수 기록의 시계가 산수로 맞는다) 흑의 시계를 9초 전으로 되돌린다
    #  → 다음 조회가 깃발을 신고한다. 수 기록을 손으로 넣으면 피셔 산수(남은 − 쓴 + 가산)가 거짓이 된다.
    seed(M_TIMEOUT, A, B, "rnbqkbnr/pppppppp/8/8/8/8/PPPPPPPP/RNBQKBNR w KQkq - 0 1", 0,
         300000, 300000, 2000, 2000, 1200)
    r = call(A, "public.chess_move(1, '%s', 0, 'e2e4')" % M_TIMEOUT, end="commit")
    assert r["status"] == "ok", r
    admin("update public.chess_matches set black_ms_left = 1000, black_grace_ms = 2000, "
          "turn_started_at = clock_timestamp() - interval '9 seconds' where id = '%s';" % M_TIMEOUT)
    call(B, "public.chess_state(1, '%s', 0)" % M_TIMEOUT, end="commit",
         name="chess_state__end_timeout", flagged="black",
         note="흑이 남은 1,000ms + 유예 2,000ms 를 9초째에 넘겼다 → white_win/timeout. "
              "조회가 깃발을 신고한다(상대가 창을 닫아도 판이 끝난다) — 지는 쪽 시점")

    # FIDE 6.9: 깃발이 떨어졌지만 시간이 남은 쪽에 메이트할 기물이 없다 → 무승부
    seed(M_TIMEOUT69, A, B, "4k3/8/8/8/8/8/8/3QK3 w - - 0 60", 118, 1000, 274000, 2000, 2000, 9000)
    call(A, "public.chess_state(1, '%s', 0)" % M_TIMEOUT69, end="commit",
         name="chess_state__end_timeout_insufficient", flagged="white",
         note="백의 깃발이 떨어졌는데 **흑은 맨 왕**이다 → FIDE 6.9 로 draw/timeout_insufficient. "
              "백이 퀸을 들고 있어 사국(chess__insufficient)은 거짓이다 — 한쪽만 보는 술어라야 무승부가 선다")

    # 무승부 합의
    seed(M_AGREE, A, B, "rnbqkbnr/pppppppp/8/8/8/8/PPPPPPPP/RNBQKBNR w KQkq - 0 1", 0,
         300000, 300000, 2000, 2000, 100)
    admin("update public.chess_matches set draw_offer_by = '%s' where id = '%s';" % (A, M_AGREE))
    call(B, "public.chess_respond_draw(1, '%s', true)" % M_AGREE, end="commit",
         name="chess_respond_draw__accept_ok", note="draw/agreement — 각자 환불")

    # 시계 역주행(차례 시작이 미래) → retry{clock_skew}
    seed(M_SKEW, A, B, "rnbqkbnr/pppppppp/8/8/8/8/PPPPPPPP/RNBQKBNR w KQkq - 0 1", 0,
         300000, 300000, 2000, 2000, -5000)
    call(A, "public.chess_move(1, '%s', 0, 'e2e4')" % M_SKEW, name="chess_move__retry_clock_skew",
         note="도착 시각이 차례 시작보다 앞 — 시간패가 아니고 수도 받지 않는다")
    admin("delete from public.chess_matches where id = '%s';" % M_SKEW)

    # 관전 꼬리 밖(오래 전에 끝난 판)
    seed(M_OLD, A, D, "R5k1/5ppp/8/8/8/8/8/6K1 b - - 1 31", 61, 120000, 90000, 2000, 2000, 0,
         status="finished", result="white_win", reason="checkmate", winner=A, finished_ago="2 days")
    call(C, "public.chess_watch(1, '%s', 0)" % M_OLD, name="chess_watch__not_found_old",
         note="끝난 지 180초를 넘긴 판 — 꼬리 밖")
    call(C, "public.chess_watch(1, '%s', 0)" % "00000000-0000-4000-8000-000000000000",
         name="chess_watch__not_found", note="없는 판")

    # 관전용으로 하나는 진행 중으로 남긴다
    seed(M_WATCH, A, B, mid_fen, 6, 281234, 274567, 2000, 2000, 1500)
    call(C, "public.chess_watch(1, '%s', 0)" % M_WATCH, name="chess_watch__ok_seeded",
         note="심은 진행 중 판 — 시계 숫자가 초기값과 다르다(기준선이 갈린다)")

    # ★ 체크·승격 — **기준선을 갈라 주는** 자리다. 이 둘이 없으면 in_check 는 모든 픽스처에서 false/null 이고
    #   promo 는 모든 수에서 null 이라, "서버가 준 값과 Swift 가 계산한 값이 같다" 가 영원히 초록이 된다.
    seed(M_CHECK, A, B, "4k3/8/8/8/8/8/4r3/4K3 w - - 10 30", 58, 240000, 237000, 2000, 2000, 1200)
    call(A, "public.chess_state(1, '%s', 0)" % M_CHECK, name="chess_state__ok_in_check",
         note="흑 룩 e2 가 백 킹 e1 을 체크 — in_check true · legal_moves 는 킹 수 + Kxe2")
    call(C, "public.chess_watch(1, '%s', 0)" % M_CHECK, name="chess_watch__ok_in_check",
         note="관전도 in_check 를 싣는다(차례인 쪽 기준 — 보는 사람 기준이 아니다)")
    seed(M_PROMO, A, B, "4k3/P7/8/8/8/8/8/4K3 w - - 0 50", 98, 210000, 205000, 2000, 2000, 900)
    call(A, "public.chess_move(1, '%s', 98, 'a7a8q')" % M_PROMO, end="commit",
         name="chess_move__ok_promotion",
         note="a8=Q+ — promo 가 null 이 아닌 유일한 수 기록(from 48(a7) · to 56(a8))")
    admin("delete from public.chess_moves where match_id in ('%s', '%s');"
          "delete from public.chess_matches where id in ('%s', '%s');" % (M_CHECK, M_PROMO, M_CHECK, M_PROMO))

    # ── 7. 순위표 ──────────────────────────────────────────────────────────────────────────
    call(A, "public.chess_ranking(1)", name="chess_ranking__ok_me_ranked", note="A 는 순위 안")
    call(C, "public.chess_ranking(1)", name="chess_ranking__ok_me_unranked",
         note="C 는 0판 — rows 는 있고 me.rank 는 null(0판 제외)")
    call(HIDDEN, "public.chess_ranking(1)", name="chess_ranking__ok_hidden_empty",
         note="숨김 계정에게는 일반 사용자 행이 안 보인다 — rows []")

    with open(os.path.join(OUT, "_manifest.json"), "w", encoding="utf-8") as fh:
        json.dump(OrderedDict(users=ids, seeds=seeds, flow_m1=flow, fixtures=manifest),
                  fh, ensure_ascii=False, indent=2)
        fh.write("\n")
    print("\n%d fixtures -> %s" % (len(manifest), OUT))
