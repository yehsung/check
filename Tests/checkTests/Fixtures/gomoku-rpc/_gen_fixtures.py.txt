def psql(sql, user="migrator"):
    p = subprocess.run([PSQL, "-X", "-q", "-At", "-v", "ON_ERROR_STOP=1", "-U", user, "-d", "postgres"],
                       input=sql, capture_output=True, text=True, env=env)
    if p.returncode != 0:
        sys.stderr.write(sql + "\n--- stderr ---\n" + p.stderr)
        raise SystemExit(f"psql failed rc={p.returncode}")
    return p.stdout


def call(caller, rpc_sql, setup="", end="rollback", name=None, note=""):
    sql = "set role postgres;\nbegin;\n" + setup + "\n"
    sql += f"select harness.login('{caller}') \\g /dev/null\n"
    sql += f"select '@@' || ({rpc_sql})::text;\n{end};\n"
    out = psql(sql)
    lines = [l for l in out.splitlines() if l.startswith("@@")]
    assert len(lines) == 1, out
    obj = json.loads(lines[0][2:], object_pairs_hook=OrderedDict)
    if name:
        with open(os.path.join(OUT, name + ".json"), "w", encoding="utf-8") as fh:
            json.dump(obj, fh, ensure_ascii=False, indent=2)
            fh.write("\n")
        manifest[name] = OrderedDict(caller=caller, caller_label=LABEL.get(caller), rpc=rpc_sql,
                                     setup=setup.strip() or None, end=end, note=note or None)
        print(f"{name:46s} status={obj.get('status')}")
    return obj


def admin(sql):
    psql("set role postgres;\nbegin;\n" + sql + "\ncommit;\n")


def set_ruby_sql(uid, target):
    return f"select contract_t.set_ruby('{uid}', {target});"


def pair(challenger, opponent, stake):
    r = call(challenger, f"public.gomoku_challenge(1, '{opponent}', {stake})", end="commit")
    assert r["status"] == "ok", r
    m = r["match_id"]
    r = call(opponent, f"public.gomoku_respond(1, '{m}', true)", end="commit")
    assert r["status"] == "ok", r
    return m, r["match"]["black"], r["match"]["white"]


def move(caller, m, seq, x, y, name=None, end="commit", setup=""):
    return call(caller, f"public.gomoku_move(1, '{m}', {seq}, {x}, {y})", end=end, name=name, setup=setup)


def drop_match(m):
    admin(f"delete from public.gomoku_matches where id = '{m}';")
