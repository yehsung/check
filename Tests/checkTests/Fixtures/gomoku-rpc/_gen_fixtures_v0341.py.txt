def psql(sql, user="migrator"):
    p = subprocess.run([PSQL, "-X", "-q", "-At", "-v", "ON_ERROR_STOP=1", "-U", user, "-d", "postgres"],
                       input=sql, capture_output=True, text=True, env=env)
    if p.returncode != 0:
        sys.stderr.write(sql + "\n--- stderr ---\n" + p.stderr)
        raise SystemExit(f"psql failed rc={p.returncode}")
    return p.stdout


def call(caller, rpc_sql, setup="", end="rollback", name=None, note="", role="authenticated"):
    login = f"select harness.login({'null::uuid' if caller is None else repr(caller)}, '{role}') \\g /dev/null\n"
    sql = "set role postgres;\nbegin;\n" + setup + "\n" + login
    sql += f"select '@@' || ({rpc_sql})::text;\n{end};\n"
    out = psql(sql)
    lines = [l for l in out.splitlines() if l.startswith("@@")]
    assert len(lines) == 1, out
    obj = json.loads(lines[0][2:], object_pairs_hook=OrderedDict)
    if name:
        with open(os.path.join(OUT, name + ".json"), "w", encoding="utf-8") as fh:
            json.dump(obj, fh, ensure_ascii=False, indent=2)
            fh.write("\n")
        new_manifest[name] = OrderedDict(caller=caller, caller_label=LABEL.get(caller), rpc=rpc_sql,
                                         setup=setup.strip() or None, end=end, note=note or None)
        print(f"{name:40s} status={obj.get('status')}")
    return obj


def admin(sql):
    psql("set role postgres;\nbegin;\n" + sql + "\ncommit;\n")


def finished(mid, ch, op, black, white, result, why, mc):
    winner = {"black_win": black, "white_win": white}.get(result, "null")
    winner = "null" if winner == "null" else f"'{winner}'"
    return (f"('{mid}', '{ch}', '{op}', 3, 'finished', '{black}', '{white}', repeat('.', 225), {mc}, null, null, "
            f"clock_timestamp() - interval '1 day 1 minute', clock_timestamp() - interval '1 day', "
            f"clock_timestamp() - interval '1 day', '{result}', '{why}', {winner})")
